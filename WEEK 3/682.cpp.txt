class Solution {
public:
    int calPoints(vector<string>& operations) {
        vector<int> scores;

        for (string op : operations) {

            if (op == "C") {
                scores.pop_back();
            }

            else if (op == "D") {
                int last = scores.back();
                scores.push_back(last * 2);
            }

            else if (op == "+") {
                int n = scores.size();

                int last = scores[n - 1];
                int secondLast = scores[n - 2];

                scores.push_back(last + secondLast);
            }

            else {
              
                int num = stoi(op);
                scores.push_back(num);
            }
        }

        int total = 0;

        for (int score : scores) {
            total += score;
        }

        return total;
    }
};
